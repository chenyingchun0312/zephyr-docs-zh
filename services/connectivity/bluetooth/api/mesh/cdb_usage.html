概述

配置数据库（CDB）是供 Bluetooth® Mesh 配网器设备使用的子系统。它存储和管理 Mesh 网络的相关信息，包括已配网节点、网络密钥（NetKey）和应用密钥（AppKey）。CDB 使配网器能够跟踪网络拓扑、分配地址，并系统地配置节点。

备注

CDB 的实现采用专有设计，并不遵循 Bluetooth SIG 规范。

CDB 通过 CONFIG_BT_MESH_CDB 配置选项启用，通常只在网络中的配网器设备上使用。为避免冲突，Mesh 网络中应当只有一台设备启用 CDB。

主要特性

  • 持久化存储：启用 CONFIG_BT_SETTINGS 时，所有 CDB 数据都会被持久化存储，使配网器能够在重启后恢复网络状态。

  • 地址管理：为已配网节点自动分配和跟踪单播地址。

  • 密钥管理：存储和同步网络密钥与应用密钥。

  • 节点跟踪：维护每个已配网节点的信息，包括 UUID、地址、元素数量、设备密钥（DevKey）和配置状态。

  • IV 索引管理：跟踪网络的 IV 索引和 IV 更新状态。

配置选项

以下 Kconfig 选项控制 CDB 的容量：

  • CONFIG_BT_MESH_CDB_NODE_COUNT - CDB 可处理的最大节点数。

  • CONFIG_BT_MESH_CDB_SUBNET_COUNT - CDB 可处理的最大子网数。

  • CONFIG_BT_MESH_CDB_APP_KEY_COUNT - CDB 可处理的最大应用密钥数。

其他选项：

  • CONFIG_BT_MESH_CDB_KEY_SYNC - 启用配网器节点上的 Mesh 密钥与 CDB 密钥之间的自动同步。启用该选项后，通过 配置服务器 对密钥执行的操作会自动同步到 CDB。

数据结构

CDB 设计为使用若干数据结构来处理收集到的数据。这些结构体分别表示节点、子网、应用密钥以及数据库的整体状态。

节点结构体

struct bt_mesh_cdb_node {
	/** Node UUID */
	uint8_t  uuid[16];
	/** Primary element address */
	uint16_t addr;
	/** Network key index used during provisioning */
	uint16_t net_idx;
	/** Number of elements */
	uint8_t  num_elem;
	/** Device key */
	struct bt_mesh_key dev_key;
	/** Node flags */
	ATOMIC_DEFINE(flags, BT_MESH_CDB_NODE_FLAG_COUNT);
};

节点标志：

  • BT_MESH_CDB_NODE_CONFIGURED - 节点已配置好密钥和绑定时设置。

子网结构体

struct bt_mesh_cdb_subnet {
	/** Network key index */
	uint16_t net_idx;
	/** Key Refresh phase */
	uint8_t kr_phase;
	/** Old and new keys for Key Refresh */
	struct {
		struct bt_mesh_key net_key;
	} keys[2];
};

应用密钥结构体

struct bt_mesh_cdb_app_key {
	/** Bound network key index */
	uint16_t net_idx;
	/** Application key index */
	uint16_t app_idx;
	/** Old and new keys for Key Refresh */
	struct {
		struct bt_mesh_key app_key;
	} keys[2];
};

CDB 主结构体

struct bt_mesh_cdb {
	/** Network IV Index */
	uint32_t iv_index;
	/** Lowest available unicast address for provisioning */
	uint16_t lowest_avail_addr;
	/** Flags for CDB state (private) */
	ATOMIC_DEFINE(flags, BT_MESH_CDB_FLAG_COUNT);
	/** Nodes in the mesh network */
	struct bt_mesh_cdb_node nodes[NODE_COUNT];
	/** Subnets in the mesh network */
	struct bt_mesh_cdb_subnet subnets[SUBNET_COUNT];
	/** Application keys in the mesh network */
	struct bt_mesh_cdb_app_key app_keys[APP_KEY_COUNT];
};

/** The global CDB instance is accessible through this variable. */
extern struct bt_mesh_cdb bt_mesh_cdb;

配网器初始化

在嵌入式配网器节点上，必须在 Mesh 协议栈完成配网之前初始化 CDB，应用程序才能使用它进行密钥存储、节点跟踪和配置。这一步会建立主网络密钥、初始子网以及起始 IV 索引和地址空间，使之后所有 CDB 操作都在一个明确定义的网络上下文中进行，并且可以在重启后持久保存。

典型的配网器在启动期间初始化 CDB：

#include <zephyr/bluetooth/mesh.h>

static int provisioner_init(void)
{
    uint8_t net_key[16];
    int err;

    /* Initialize Bluetooth Mesh */
    err = bt_mesh_init(&prov, &comp);
    if (err) {
        return err;
    }

    /* Load settings if persistent storage is enabled */
    if (IS_ENABLED(CONFIG_BT_SETTINGS)) {
        settings_load();
    }

    /* Generate or use predefined network key */
    bt_rand(net_key, 16);

    /* Create CDB with primary network key */
    err = bt_mesh_cdb_create(net_key);
    if (err == -EALREADY) {
        printk("Using stored CDB\n");
    } else if (err) {
        printk("Failed to create CDB (err %d)\n", err);
        return err;
    } else {
        printk("Created new CDB\n");
    }

    return 0;
}

要点：

  • 使用主网络密钥调用 bt_mesh_cdb_create() 函数。

  • 如果 CDB 已存在（从持久化存储中加载），则返回错误码。

  • 该函数会自动创建一个 NetIdx = 0 的子网。

  • 将 IV 索引设置为 0，将最低可用地址设置为 1。

配网与节点分配

向 Mesh 中添加新设备时，必须在配网器的数据库中一致地记录它们的地址、设备密钥和基本元数据。在 PB-ADV/PB-GATT 配网期间自动分配节点，可以使 CDB 与实际网络保持同步。为新设备配网时，CDB 会在配网过程中自动分配节点。

不过，也可以手动分配节点或检查分配情况。手动分配节点适用于高级场景，例如从外部来源导入节点、预先分配地址范围，或者根据已知信息恢复网络。

/* Provisioning callback - node is automatically added to CDB */
static void node_added(uint16_t idx, uint8_t uuid[16], uint16_t addr,
                       uint8_t num_elem)
{
    printk("Node added: addr=0x%04x, elements=%d\n", addr, num_elem);
    /* The CDB node is created automatically by the provisioning subsystem */
}

static const struct bt_mesh_prov prov = {
    .uuid = dev_uuid,
    .node_added = node_added,
    /* ... other callbacks ... */
};

/* Manual node allocation (if needed) */
static struct bt_mesh_cdb_node *allocate_node(const uint8_t uuid[16],
                                                uint8_t num_elem)
{
    struct bt_mesh_cdb_node *node;
    uint16_t addr;

    /* Get free address or specify one (0 = auto-allocate) */
    addr = bt_mesh_cdb_free_addr_get(num_elem);
    if (addr == BT_MESH_ADDR_UNASSIGNED) {
        printk("No free addresses available\n");
        return NULL;
    }

    /* Allocate node in CDB */
    node = bt_mesh_cdb_node_alloc(uuid, addr, num_elem, net_idx);
    if (node == NULL) {
        printk("Failed to allocate node\n");
        return NULL;
    }

    return node;
}

要点：

  • 在正常的 PB-ADV/PB-GATT 配网期间，节点会被自动添加到 CDB 中。

  • bt_mesh_cdb_node_alloc() 函数使用指定的参数创建一个 CDB 条目。

  • 传入 BT_MESH_ADDR_UNASSIGNED 作为地址，即可让 CDB 自动分配最低的可用地址。

  • bt_mesh_cdb_free_addr_get() 函数为给定的元素数量查找一个空闲的地址范围。

  • 地址分配会检查冲突并确保单播地址有效。

子网与密钥管理

大型或分段的 Mesh 网络通常使用多个子网来隔离流量域、实现安全迁移或支持密钥刷新（Key Refresh）过程。在 CDB 中管理子网及其网络密钥，使配网器能够创建新子网、轮换密钥，并在为节点配网和配置节点时得出正确的标志和 IV 索引。

在 CDB 中管理网络密钥和子网：

/* Add a new subnet */
static int add_subnet(uint16_t net_idx)
{
    struct bt_mesh_cdb_subnet *sub;
    uint8_t net_key[16];
    int err;

    /* Allocate subnet in CDB */
    sub = bt_mesh_cdb_subnet_alloc(net_idx);
    if (sub == NULL) {
        printk("Failed to allocate subnet\n");
        return -ENOMEM;
    }

    /* Generate network key */
    bt_rand(net_key, 16);

    /* Import key value */
    err = bt_mesh_cdb_subnet_key_import(sub, 0, net_key);
    if (err) {
        bt_mesh_cdb_subnet_del(sub, false);
        return err;
    }

    /* Store subnet */
    if (IS_ENABLED(CONFIG_BT_SETTINGS)) {
        bt_mesh_cdb_subnet_store(sub);
    }

    return 0;
}

/* Get subnet for provisioning */
static int get_subnet_flags(uint16_t net_idx)
{
    struct bt_mesh_cdb_subnet *sub;
    uint8_t flags;

    sub = bt_mesh_cdb_subnet_get(net_idx);
    if (sub == NULL) {
        return -ENOENT;
    }

    flags = bt_mesh_cdb_subnet_flags(sub);
    printk("Subnet flags: KR=%d IVU=%d\n",
           !!(flags & BT_MESH_NET_FLAG_KR),
           !!(flags & BT_MESH_NET_FLAG_IVU));

    return 0;
}

要点：

  • bt_mesh_cdb_subnet_alloc() 函数按 NetIdx 分配子网。

  • bt_mesh_cdb_subnet_key_import() 函数设置实际的密钥值。

  • bt_mesh_cdb_subnet_flags() 函数返回配网数据所需的标志。

  • 这些标志包括来自 CDB 的密钥刷新和 IV 更新状态。

子网查找与移除

有时可能需要检查某个现有子网（用于诊断或工具），或者将其完全移除（例如停用某个网段、轮换到新子网，或在实验后进行清理）。CDB 提供了查找和删除辅助函数，使配网器所掌握的活动子网信息能够与节点实际使用的情况保持一致。

在 CDB 中查找和管理子网：

/* Get subnet by NetIdx */
static void lookup_subnet(uint16_t net_idx)
{
    struct bt_mesh_cdb_subnet *sub;

    sub = bt_mesh_cdb_subnet_get(net_idx);
    if (sub == NULL) {
        printk("Subnet 0x%03x not found\n", net_idx);
        return;
    }

    printk("Subnet 0x%03x: kr_phase=%d\n", net_idx, sub->kr_phase);
}

/* Remove a subnet from the CDB */
static void remove_subnet(uint16_t net_idx)
{
    struct bt_mesh_cdb_subnet *sub;

    sub = bt_mesh_cdb_subnet_get(net_idx);
    if (sub == NULL) {
        printk("Subnet not found\n");
        return;
    }

    /* First, remove the subnet from all nodes in the network */
    /* ... send NetKey Delete to all nodes using Config Client ... */

    /* Delete subnet from CDB */
    bt_mesh_cdb_subnet_del(sub, true);

    printk("Subnet 0x%03x removed\n", net_idx);
}

/* Export subnet key for external use */
static int export_subnet_key(uint16_t net_idx, int key_idx)
{
    struct bt_mesh_cdb_subnet *sub;
    uint8_t net_key[16];
    int err;

    sub = bt_mesh_cdb_subnet_get(net_idx);
    if (sub == NULL) {
        return -ENOENT;
    }

    err = bt_mesh_cdb_subnet_key_export(sub, key_idx, net_key);
    if (err) {
        printk("Failed to export subnet key (err %d)\n", err);
        return err;
    }

    printk("NetKey[%d] for subnet 0x%03x: ", key_idx, net_idx);
    for (int i = 0; i < 16; i++) {
        printk("%02x", net_key[i]);
    }
    printk("\n");

    return 0;
}

要点：

  • bt_mesh_cdb_subnet_get() 函数按 NetKeyIndex 获取子网。

  • 向 bt_mesh_cdb_subnet_del() 函数传入 true 以清除持久化存储。

  • 在从 CDB 中删除子网之前，务必先将其从所有节点上移除。

  • 使用 bt_mesh_cdb_subnet_key_export() 函数安全地获取密钥材料。

  • 在密钥刷新过程中，key_idx 参数（0 或 1）用于在旧密钥和新密钥之间进行选择。

设置应用密钥

应用密钥决定了哪些应用流量可以被加密，以及可以绑定到哪里。CDB 初始化后，配网器必须创建一个或多个应用密钥，以便在节点上绑定模型，从而实现真正的应用层通信（例如照明或传感器数据）。

创建 CDB 后，添加用于节点配置的应用密钥：

static void setup_cdb_keys(void)
{
    struct bt_mesh_cdb_app_key *key;
    uint8_t app_key[16];
    int err;

    /* Allocate application key in CDB */
    key = bt_mesh_cdb_app_key_alloc(net_idx, app_idx);
    if (key == NULL) {
        printk("Failed to allocate app-key\n");
        return;
    }

    /* Generate random key value */
    bt_rand(app_key, 16);

    /* Import the key into CDB */
    err = bt_mesh_cdb_app_key_import(key, 0, app_key);
    if (err) {
        printk("Failed to import appkey (err %d)\n", err);
        return;
    }

    /* Store to persistent storage */
    if (IS_ENABLED(CONFIG_BT_SETTINGS)) {
        bt_mesh_cdb_app_key_store(key);
    }
}

要点：

  • bt_mesh_cdb_app_key_alloc() 函数在 CDB 中分配一个槽位。

  • bt_mesh_cdb_app_key_import() 函数设置实际的密钥值。

  • 导入函数的第二个参数（0）是用于密钥刷新过程的密钥索引（0 = 当前密钥）。

  • 使用持久化存储时，务必在创建密钥后将其存储。

应用密钥查找与移除

随着时间推移，您可能需要查看某个应用密钥绑定到了哪个网络密钥、轮换或吊销应用密钥，或者清理不再使用的密钥。通过 CDB 正确地查找和移除密钥，有助于避免悬空的绑定，并确保节点和配网器对可用应用密钥的认识保持一致。

在 CDB 中查找和管理应用密钥：

/* Get application key by AppIdx */
static void lookup_app_key(uint16_t app_idx)
{
    struct bt_mesh_cdb_app_key *key;

    key = bt_mesh_cdb_app_key_get(app_idx);
    if (key == NULL) {
        printk("AppKey 0x%03x not found\n", app_idx);
        return;
    }

    printk("AppKey 0x%03x: bound to NetIdx 0x%03x\n",
           app_idx, key->net_idx);
}

/* Remove an application key from the CDB */
static void remove_app_key(uint16_t app_idx)
{
    struct bt_mesh_cdb_app_key *key;

    key = bt_mesh_cdb_app_key_get(app_idx);
    if (key == NULL) {
        printk("AppKey not found\n");
        return;
    }

    /* First, remove the app key from all nodes in the network */
    /* ... send AppKey Delete to all nodes using Config Client ... */

    /* Delete app key from CDB */
    bt_mesh_cdb_app_key_del(key, true);

    printk("AppKey 0x%03x removed\n", app_idx);
}

/* Export application key for external use */
static int export_app_key(uint16_t app_idx, int key_idx)
{
    struct bt_mesh_cdb_app_key *key;
    uint8_t app_key[16];
    int err;

    key = bt_mesh_cdb_app_key_get(app_idx);
    if (key == NULL) {
        return -ENOENT;
    }

    err = bt_mesh_cdb_app_key_export(key, key_idx, app_key);
    if (err) {
        printk("Failed to export app key (err %d)\n", err);
        return err;
    }

    printk("AppKey[%d] 0x%03x: ", key_idx, app_idx);
    for (int i = 0; i < 16; i++) {
        printk("%02x", app_key[i]);
    }
    printk("\n");

    return 0;
}

/* Update application key binding */
static int update_app_key_binding(uint16_t app_idx, uint16_t new_net_idx)
{
    struct bt_mesh_cdb_app_key *key;

    key = bt_mesh_cdb_app_key_get(app_idx);
    if (key == NULL) {
        return -ENOENT;
    }

    /* Verify the new subnet exists */
    if (bt_mesh_cdb_subnet_get(new_net_idx) == NULL) {
        printk("Target subnet 0x%03x not found\n", new_net_idx);
        return -ENOENT;
    }

    key->net_idx = new_net_idx;

    /* Store updated binding */
    if (IS_ENABLED(CONFIG_BT_SETTINGS)) {
        bt_mesh_cdb_app_key_store(key);
    }

    printk("AppKey 0x%03x rebound to NetIdx 0x%03x\n",
           app_idx, new_net_idx);

    return 0;
}

要点：

  • bt_mesh_cdb_app_key_get() 函数按 AppKeyIndex 获取应用密钥。

  • 向 bt_mesh_cdb_app_key_del() 函数传入 true 以清除持久化存储。

  • 在从 CDB 中删除应用密钥之前，务必先将其从所有节点上移除。

  • 使用 bt_mesh_cdb_app_key_export() 函数安全地获取密钥材料。

  • 在密钥刷新过程中，key_idx 参数（0 或 1）用于在旧密钥和新密钥之间进行选择。

  • 应用密钥通过 net_idx 字段绑定到某个网络密钥。

节点配置

配网只是将节点加入网络，并不会使其参与到您的应用中。配网之后，您必须配置节点：添加网络密钥/应用密钥、绑定模型，以及设置订阅/发布。CDB 存储密钥和配置状态，使配网器能够在重启后继续配置，并避免重复配置已完成配置的节点。

配网之后，通过添加密钥和绑定模型来配置节点：

static void configure_node(struct bt_mesh_cdb_node *node)
{
    NET_BUF_SIMPLE_DEFINE(buf, BT_MESH_RX_SDU_MAX);
    struct bt_mesh_cdb_app_key *key;
    uint8_t app_key[16];
    uint8_t status;
    int err;

    printk("Configuring node 0x%04x\n", node->addr);

    /* Get application key from CDB */
    key = bt_mesh_cdb_app_key_get(app_idx);
    if (key == NULL) {
        printk("App-key not found\n");
        return;
    }

    /* Export key value from CDB */
    err = bt_mesh_cdb_app_key_export(key, 0, app_key);
    if (err) {
        printk("Failed to export key (err %d)\n", err);
        return;
    }

    /* Add application key to the node */
    err = bt_mesh_cfg_cli_app_key_add(net_idx, node->addr,
                                      net_idx, app_idx,
                                      app_key, &status);
    if (err || status) {
        printk("Failed to add app-key (err %d, status %d)\n",
               err, status);
        return;
    }

    /* Get composition data and bind models */
    err = bt_mesh_cfg_cli_comp_data_get(net_idx, node->addr, 0,
                                        &status, &buf);
    if (err || status) {
        printk("Failed to get composition data\n");
        return;
    }

    /* Parse composition and bind models to app key */
    /* ... model binding code ... */

    /* Mark node as configured */
    atomic_set_bit(node->flags, BT_MESH_CDB_NODE_CONFIGURED);

    /* Persist configuration status */
    if (IS_ENABLED(CONFIG_BT_SETTINGS)) {
        bt_mesh_cdb_node_store(node);
    }
}

要点：

  • 使用 bt_mesh_cdb_app_key_get() 和 bt_mesh_cdb_app_key_export() 函数获取密钥。

  • 配置客户端模型用于配置远程节点。

  • 配置成功后设置 BT_MESH_CDB_NODE_CONFIGURED 标志。

  • 务必调用 bt_mesh_cdb_node_store() 函数持久化保存已配置状态。

遍历节点

许多管理任务都需要对所有节点（或经过筛选的部分节点）进行操作：检查哪些节点仍需配置、生成统计信息或执行批量操作。CDB 迭代器让配网器能够以安全、抽象的方式遍历所有已知节点，而无需依赖内部存储细节。

CDB 提供了一个迭代器来处理所有已分配的节点：

/* Check for unconfigured nodes */
static uint8_t check_unconfigured(struct bt_mesh_cdb_node *node,
                                   void *user_data)
{
    if (!atomic_test_bit(node->flags, BT_MESH_CDB_NODE_CONFIGURED)) {
        printk("Node 0x%04x needs configuration\n", node->addr);
        configure_node(node);
    }

    return BT_MESH_CDB_ITER_CONTINUE;
}

static void process_nodes(void)
{
    bt_mesh_cdb_node_foreach(check_unconfigured, NULL);
}

/* Example: Count nodes on a specific subnet */
static uint8_t count_subnet_nodes(struct bt_mesh_cdb_node *node,
                                   void *user_data)
{
    uint16_t *net_idx = user_data;
    static int count = 0;

    if (node->net_idx == *net_idx) {
        count++;
    }

    return BT_MESH_CDB_ITER_CONTINUE;
}

要点：

  • bt_mesh_cdb_node_foreach() 函数为每个已分配的节点调用回调。

  • 返回 BT_MESH_CDB_ITER_CONTINUE 以继续迭代。

  • 返回 BT_MESH_CDB_ITER_STOP 以提前停止迭代。

  • 只有 addr != BT_MESH_ADDR_UNASSIGNED 的节点才会调用该回调。

节点查找与移除

有时需要检查单个节点（例如在调试时），或者将其从网络中移除（例如在物理上停用或更换该节点时）。使用 CDB 的节点查找和删除功能，可以使 Mesh 的逻辑视图与实际设备保持一致，并防止地址重用冲突。

在 CDB 中查找和管理节点：

/* Get node by address */
static void lookup_node(uint16_t addr)
{
    struct bt_mesh_cdb_node *node;

    node = bt_mesh_cdb_node_get(addr);
    if (node == NULL) {
        printk("Node 0x%04x not found\n", addr);
        return;
    }

    printk("Node 0x%04x: %d elements, net_idx=0x%03x\n",
           node->addr, node->num_elem, node->net_idx);
}

/* Remove a node from the network */
static void remove_node(uint16_t addr)
{
    struct bt_mesh_cdb_node *node;

    node = bt_mesh_cdb_node_get(addr);
    if (node == NULL) {
        printk("Node not found\n");
        return;
    }

    /* First, send Node Reset to the device (recommended) */
    /* ... send reset command using Config Client ... */

    /* Delete node from CDB */
    bt_mesh_cdb_node_del(node, true);

    printk("Node 0x%04x removed\n", addr);
}

要点：

  • bt_mesh_cdb_node_get() 函数按元素地址进行搜索（适用于任意元素地址）。

  • 在移除节点之前，务必先发送 Node Reset 命令，以免产生孤立设备。

  • 向 bt_mesh_cdb_node_del() 函数传入 true 以清除持久化存储。

  • 删除节点可能会在适当情况下更新 lowest_avail_addr（详情请阅读 CDB API）。

IV 索引管理

IV 索引是 Bluetooth Mesh 安全和重放保护的核心组成部分。发生 IV 更新过程时，配网器必须更新其存储的 IV 索引，以便之后的配网和配置使用正确的值和地址空间。将其存储在 CDB 中，可以使网络状态在重启之间以及各次配网操作之间保持一致。

更新网络的 IV 索引：

/* Update IV Index when IV Update procedure occurs */
static void handle_iv_update(uint32_t iv_index, bool iv_update)
{
    bt_mesh_cdb_iv_update(iv_index, iv_update);

    printk("IV Index updated: %u (IV Update: %s)\n",
           iv_index, iv_update ? "in progress" : "normal");
}

/* The IV Index is automatically used during provisioning */
static void provision_with_current_iv(void)
{
    /* The provisioner automatically uses bt_mesh_cdb.iv_index
     * and bt_mesh_cdb_subnet_flags() when sending provisioning data
     */
}

要点：

  • bt_mesh_cdb_iv_update() 函数同时更新 IV 索引和 IV 更新标志。

  • 在 IV 索引更新期间，CDB 会自动重置 lowest_avail_addr。

  • 配网器子系统会自动为新节点使用 CDB 的 IV 索引。

使用设备密钥

设备密钥是敏感信息，通常通过安全密钥存储（例如 PSA crypto）进行处理。CDB 的密钥导入/导出 API 对密钥的存储方式进行了抽象，使应用程序能够在需要时（用于配置或外部工具）使用设备密钥，而不会破坏安全模型，也不依赖指向密钥存储的直接指针。

使用密钥管理 API 导入和导出设备密钥：

/* Import a known device key */
static int import_dev_key(struct bt_mesh_cdb_node *node,
                          const uint8_t dev_key[16])
{
    int err;

    err = bt_mesh_cdb_node_key_import(node, dev_key);
    if (err) {
        printk("Failed to import device key (err %d)\n", err);
        return err;
    }

    /* Store updated node */
    if (IS_ENABLED(CONFIG_BT_SETTINGS)) {
        bt_mesh_cdb_node_store(node);
    }

    return 0;
}

/* Export device key for external use */
static int export_dev_key(struct bt_mesh_cdb_node *node)
{
    uint8_t dev_key[16];
    int err;

    err = bt_mesh_cdb_node_key_export(node, dev_key);
    if (err) {
        printk("Failed to export device key (err %d)\n", err);
        return err;
    }

    /* Use the exported key */
    printk("Device key: ");
    for (int i = 0; i < 16; i++) {
        printk("%02x", dev_key[i]);
    }
    printk("\n");

    return 0;
}

要点：

  • 处理密钥时务必使用导入/导出函数（使用 PSA crypto 时必须如此）。

  • 密钥被安全存储，无法通过指针直接访问。

  • 配网器在配网期间会自动生成并导入设备密钥。

清空 CDB

在恢复出厂设置、测试或建立一个全新网络时，您可能需要清除所有配网和配置数据。清空 CDB 会重置逻辑网络状态，以便您可以安全地从头重建，同时避免与过时的密钥或节点条目产生不一致。

重置整个配置数据库：

static void reset_network(void)
{
    printk("Clearing CDB...\n");

    /* Clear all CDB data */
    bt_mesh_cdb_clear();

    printk("CDB cleared\n");
}

要点：

  • bt_mesh_cdb_clear() 函数移除所有节点、子网和应用密钥。

  • 如果启用了 CONFIG_BT_SETTINGS Kconfig 选项，还会清除持久化存储。

  • 将 CDB 标记为无效（必须调用 bt_mesh_cdb_create() 函数才能再次使用）。

最佳实践

  • 错误处理 - 检查所有 CDB 函数的返回值，尤其是分配函数，它们在达到容量上限时可能会失败。

  • 地址管理 - 除非有特定的寻址要求，否则请向 bt_mesh_cdb_node_alloc() 函数传入 0，让 CDB 管理地址分配。

  • 配置跟踪 - 配置成功后，务必设置 BT_MESH_CDB_NODE_CONFIGURED 标志并存储节点，以避免不必要地重新配置节点。

  • 密钥安全 - 切勿在生产代码中记录或暴露密钥。使用导入/导出函数安全地处理密钥材料。

  • 迭代安全 - 不要在 bt_mesh_cdb_node_foreach() 迭代期间修改 CDB（添加/移除节点）。应先收集地址，再进行修改。

常见陷阱

多个配网器之间的 CDB 同步

如果网络中有多个配网器，请确保它们同步各自的 CDB 以避免冲突。请使用应用层机制在多个配网器之间共享 CDB。CDB 没有内置的同步机制。

忘记存储

对 CDB 的修改不会自动持久化。使用持久化存储时，务必在修改后调用相应的 _store() 函数。

密钥索引错误

导入/导出密钥时，密钥索引参数（0 或 1）指的是在密钥刷新（Key Refresh）密钥数组中的位置，而不是 NetIdx 或 AppIdx。

地址冲突

如果手动指定地址，请确保它们不会与现有节点冲突。使用 bt_mesh_cdb_free_addr_get() 函数查找可用地址。

超出容量

CDB 具有由 Kconfig 定义的固定容量。超出容量会导致分配失败。请监控 CDB 的使用情况，并在需要时增加容量。

访问无效节点

在解引用之前，务必检查 _get() 或 _alloc() 函数返回的指针是否为 NULL。
