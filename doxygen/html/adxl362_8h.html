 禁用 FIFO
#define ADXL362_FIFO_MODE_OLDEST_SAVED   0x1
 满时停止，保留最旧的样本
#define ADXL362_FIFO_MODE_STREAM   0x2
 连续流，覆盖最旧的样本
#define ADXL362_FIFO_MODE_TRIGGERED   0x3
 在触发事件前后采集样本
