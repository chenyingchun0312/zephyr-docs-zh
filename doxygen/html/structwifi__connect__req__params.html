const uint8_t * anon_id
 用于 Wi-Fi 企业级网络 EAP 认证（阶段 1）的匿名身份。在 PEAP 或 TTLS 等 EAP 方法中，匿名身份在初始外部认证交换中发送，以保护用户的真实身份（eap_identity）。此值并非总是必需；如果未设置，真实身份可能会暴露。仅在阶段 1（外部认证）中使用。

const uint8_t * key_passwd
 用于 Wi-Fi 企业级认证的私钥密码。在需要客户端证书和私钥时使用（例如 EAP-TLS）。这是保护私钥文件的密码。使用客户端证书时适用于阶段 1（外部认证）。
uint8_t key_passwd_length
 私钥密码的长度，最大 128 字节。
const uint8_t * key2_passwd
 次要私钥的密码，如果认证方法需要的话。很少使用；通常只在高级企业级配置中需要。如果使用第二个私钥，则适用于阶段 1（外部认证）。
uint8_t key2_passwd_length
 次要私钥密码的长度，最大 128 字节。
enum wifi_wpa3_enterprise_type wpa3_ent_mode
 WPA3 企业级模式类型。选择要使用的 WPA3 企业级认证变体。WPA3 企业级是用于 Wi-Fi 网络的安全协议，常用于组织中，相比 WPA2 提供增强的安全性。此字段仅与企业级网络相关。适用于阶段 1（外部认证）。
uint8_t TLS_cipher
 用于 EAP-TLS 认证的 TLS 密码套件。此字段选择用于安全连接的加密算法。仅与使用 EAP-TLS 或类似方法的企业级网络相关。适用于阶段 1（外部认证）。
int eap_ver
 要使用的 EAP（可扩展认证协议）版本。EAP 是网络认证框架，在企业级 Wi-Fi 中常用。此字段允许在网络要求时指定协议版本。适用于阶段 1（外部认证）。值为 -1 将进行版本协商。
const uint8_t * eap_identity
 用于 EAP 认证的身份字符串（阶段 2，内部认证）。这是在安全隧道建立后（例如在 PEAP 或 TTLS 内部）呈现给认证服务器的真实用户名或身份。大多数企业级 Wi-Fi 网络（例如 WPA2/WPA3 企业级）都需要。适用于阶段 2（内部认证）。
uint8_t eap_id_length
 EAP 身份的长度，最大 64 字节。
const uint8_t * eap_password
 用于 EAP 认证的密码字符串（阶段 2，内部认证）。用于需要密码的 EAP 方法（例如 PEAP、TTLS、EAP-FAST）。这是用户用于企业级 Wi-Fi 网络的密码。适用于阶段 2（内部认证）。
uint8_t eap_passwd_length
 EAP 密码的长度，最大 128 字节。
bool verify_peer_cert
 认证期间是否验证服务器的证书颁发机构（CA）。设为 true 要求验证服务器证书（出于安全考虑推荐）。设为 false 则跳过 CA 验证（不推荐，但有时用于测试）。适用于阶段 1（外部认证）。
bool ft_used
 指示是否使用快速 BSS 转换（802.11r）。快速 BSS 转换允许在企业级网络的接入点之间无缝漫游。适用于整个连接，不特定于 EAP 阶段。
int nusers
 提供的 EAP 用户身份数量。用于需要多个用户凭据的高级企业级认证场景。适用于阶段 2（内部认证）。
uint8_t passwds
 提供的 EAP 密码数量。与企业级认证中的多个用户身份配合使用。适用于阶段 2（内部认证）。
const uint8_t * identities [WIFI_ENT_IDENTITY_MAX_USERS]
 用于 EAP 认证的用户身份字符串指针数组。用于需要多个用户身份的企业级 Wi-Fi 网络。适用于阶段 2（内部认证）。
const uint8_t * passwords [WIFI_ENT_IDENTITY_MAX_USERS]
 用于 EAP 认证的用户密码字符串指针数组。用于需要多个用户密码的企业级 Wi-Fi 网络。适用于阶段 2（内部认证）。
uint8_t ignore_broadcast_ssid
 隐藏 SSID 配置 0：禁用（默认） 1：在信标中发送空（length=0）SSID 并忽略针对广播 SSID 的探测请求 2：清除 SSID，但保留原始长度并忽略针对广播 SSID 的探测请求
