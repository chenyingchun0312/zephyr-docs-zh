宏定义

#define DT_MEM_ATTR_GET(x)
 从完整的 zephyr,memory-attr 值中提取通用内存属性位。
#define DT_MEM_CACHEABLE   BIT(0)
 内存可缓存。
#define DT_MEM_NON_VOLATILE   BIT(1)
 内存非易失。
#define DT_MEM_OOO   BIT(2)
 内存支持乱序访问。
#define DT_MEM_DMA   BIT(3)
 内存可用于 DMA。
#define DT_MEM_UNKNOWN   BIT(15)
 通用内存属性未知。
